from pathlib import Path from datetime import datetime from zoneinfo import ZoneInfo  import pandas as pd from IPython.display import display   # 1. Locate project files  project_root = Path.cwd()  if project_root.name == "notebooks":     project_root = project_root.parent  references_path = project_root / "references"  source_register_path = references_path / "source_register.csv" menu_pricing_path = (     project_root / "data" / "raw" / "menu_pricing.csv" )   # 2. Load and check saved datasets  source_register = pd.read_csv(     source_register_path,     dtype="string", )  menu_pricing = pd.read_csv(     menu_pricing_path,     dtype="string", )  if source_register["source_id"].duplicated().any():     raise ValueError("Duplicate source IDs found.")  if menu_pricing["observation_id"].duplicated().any():     raise ValueError("Duplicate observation IDs found.")  if source_register["source_id"].eq("SRC0034").any():     raise ValueError("SRC0034 already exists. No changes made.")  new_ids = ["MNU0027", "MNU0028", "MNU0029"]  if menu_pricing["observation_id"].isin(new_ids).any():     raise ValueError("Planned observation IDs already exist.")  source_columns = source_register.columns.tolist() menu_columns = menu_pricing.columns.tolist()   # 3. Capture and save the supplied browser text  pickup_text = input("Paste the complete PICKUP menu text: ")  required_terms = [     "AMERICANO",     "ICED LATTE",     "KAPE KASTILA",     "PHP 50",     "PHP 75", ]  if not all(term in pickup_text for term in required_terms):     raise ValueError(         "Expected menu details were not found. Nothing saved."     )  capture_time = datetime.now(ZoneInfo("Asia/Manila")) capture_label = capture_time.strftime("%Y%m%d_%H%M%S_%f")  pickup_evidence_path = (     references_path     / "website_evidence"     / "manual_20261023"     / f"SRC0034_pickup_menu_{capture_label}.txt" )  pickup_evidence_path.parent.mkdir(parents=True, exist_ok=True)  pickup_evidence_path.write_text(     pickup_text.strip(),     encoding="utf-8", )   # 4. Register the inspected browser evidence  new_source = pd.DataFrame([     {         "source_id": "SRC0034",         "brand_id": "BFC002",         "research_question_ids": "RQ01;RQ02;RQ10",         "source_type": (             "Official website text supplied through "             "manual browser collection"         ),         "source_title": "PICKUP COFFEE rendered menu with prices",         "source_url": "https://pickup-coffee.com/menu",         "collected_date": capture_time.strftime("%Y-%m-%d"),         "evidence_file": pickup_evidence_path.relative_to(             project_root         ).as_posix(),         "evidence_basis": "Company statement",         "verification_status": "Verified",         "access_status": "Accessible",         "historical_period": "Current",         "notes": (             "Supplied browser text inspected. Product card prices: "             "AMERICANO PHP 50, ICED LATTE PHP 75, "             "KAPE KASTILA PHP 75. Sizes and branch scope unresolved. "             "Full menu retained; only initial basket entered."         ),     } ]).reindex(columns=source_columns)   # 5. Prepare three product card observations  new_observations = pd.DataFrame(     [         [             "MNU0027",             "AMERICANO",             "Americano",             "Not stated",             "50",         ],         [             "MNU0028",             "ICED LATTE",             "Cafe latte",             "Iced",             "75",         ],         [             "MNU0029",             "KAPE KASTILA",             pd.NA,             "Not stated",             "75",         ],     ],     columns=[         "observation_id",         "product_name",         "comparison_group",         "temperature",         "displayed_price_php",     ], )  new_observations = new_observations.reindex(     columns=menu_columns ).astype("string")  new_observations["brand_id"] = "BFC002" new_observations["source_id"] = "SRC0034" new_observations["observed_date"] = "2026-10-23" new_observations["channel"] = "Official website" new_observations["location_scope"] = "Not stated in captured text" new_observations["product_category"] = "Coffee" new_observations["price_basis"] = "Product card" new_observations["availability_status"] = "Listed" new_observations["notes"] = (     "Size, volume, and branch scope unresolved. "     "Regular or promotional classification not established. "     "Hot availability does not establish a separate hot price." )  kastila_row = (     new_observations["observation_id"] == "MNU0029" )  new_observations.loc[kastila_row, "notes"] = (     "Described as condensed milk, milk, and espresso. "     "Candidate Spanish latte comparison; equivalence unresolved. "     "Size, temperature, branch scope, and price classification "     "remain unresolved." )   # 6. Append and save  source_register = pd.concat(     [source_register, new_source],     ignore_index=True, )  menu_pricing = pd.concat(     [menu_pricing, new_observations],     ignore_index=True, )  source_register.to_csv(source_register_path, index=False) menu_pricing.to_csv(menu_pricing_path, index=False)  print("Evidence saved:", pickup_evidence_path) print("Sources added:", len(new_source)) print("Observations added:", len(new_observations)) print("Total menu observations:", len(menu_pricing))  display(     menu_pricing.loc[         menu_pricing["source_id"] == "SRC0034",         [             "observation_id",             "product_name",             "comparison_group",             "temperature",             "size_label",             "displayed_price_php",             "source_id",         ],     ] )
